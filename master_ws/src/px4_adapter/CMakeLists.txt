cmake_minimum_required(VERSION 3.16)
project(px4_adapter_ros2)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(msgs REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(uav_ros2_support REQUIRED)
find_package(tinyxml2 REQUIRED)
add_executable(px4_adapter_node src/main.cpp src/adapter_node.cpp)
target_compile_features(px4_adapter_node PRIVATE cxx_std_17)
target_compile_options(px4_adapter_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(px4_adapter_node PRIVATE include)
ament_target_dependencies(px4_adapter_node rclcpp msgs px4_msgs uav_ros2_support)
target_link_libraries(px4_adapter_node tinyxml2::tinyxml2)
install(TARGETS px4_adapter_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(px4_control_test test/test_control.cpp)
  target_compile_features(px4_control_test PRIVATE cxx_std_17)
  target_include_directories(px4_control_test PRIVATE include)
  add_test(NAME px4_control COMMAND px4_control_test)
endif()
ament_package()
