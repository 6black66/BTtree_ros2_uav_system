cmake_minimum_required(VERSION 3.16)
project(flight_controller_ros2)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tinyxml2 REQUIRED)
add_library(qlet_protocol src/qlet_protocol.cpp src/udp_transport.cpp)
target_include_directories(qlet_protocol PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_options(qlet_protocol PRIVATE -Wall -Wextra -Wpedantic)
add_executable(controller_node src/main.cpp src/controller_node.cpp)
target_include_directories(controller_node PRIVATE include)
ament_target_dependencies(controller_node rclcpp msgs std_msgs)
target_link_libraries(controller_node qlet_protocol tinyxml2::tinyxml2)
target_compile_options(controller_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS controller_node qlet_protocol RUNTIME DESTINATION lib/${PROJECT_NAME} LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(qlet_protocol_test test/test_qlet_protocol.cpp)
  target_link_libraries(qlet_protocol_test qlet_protocol)
  add_test(NAME qlet_protocol COMMAND qlet_protocol_test)
endif()
ament_package()
