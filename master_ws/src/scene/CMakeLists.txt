cmake_minimum_required(VERSION 3.16)
project(video_gateway_ros2)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(msgs REQUIRED)
find_package(uav_shared_video REQUIRED)
find_package(uav_ros2_support REQUIRED)
find_package(tinyxml2 REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale)
add_library(scene_decoder src/video_decoder.cpp)
target_include_directories(scene_decoder PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_link_libraries(scene_decoder PUBLIC uav_shared_video::uav_shared_video PkgConfig::FFMPEG)
target_compile_options(scene_decoder PRIVATE -Wall -Wextra -Wpedantic)
# Retain the executable name used by the existing system launch.
add_executable(episode_node src/main.cpp src/scene_node.cpp)
target_include_directories(episode_node PRIVATE include)
ament_target_dependencies(episode_node rclcpp std_msgs msgs uav_ros2_support)
target_link_libraries(episode_node scene_decoder tinyxml2::tinyxml2 Threads::Threads)
target_compile_options(episode_node PRIVATE -Wall -Wextra -Wpedantic)
add_executable(scene_frame_consumer src/frame_consumer.cpp)
ament_target_dependencies(scene_frame_consumer rclcpp msgs)
target_link_libraries(scene_frame_consumer uav_shared_video::uav_shared_video)
install(TARGETS episode_node scene_frame_consumer scene_decoder RUNTIME DESTINATION lib/${PROJECT_NAME} LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(make_scene_test_video test/make_test_video.cpp)
  target_link_libraries(make_scene_test_video PkgConfig::FFMPEG)
  add_executable(scene_decoder_test test/test_decoder.cpp)
  target_link_libraries(scene_decoder_test scene_decoder)
  add_test(NAME make_scene_fixture COMMAND make_scene_test_video ${CMAKE_CURRENT_BINARY_DIR}/scene_fixture.ts)
  set_tests_properties(make_scene_fixture PROPERTIES FIXTURES_SETUP scene_fixture)
  add_test(NAME scene_decoder COMMAND scene_decoder_test ${CMAKE_CURRENT_BINARY_DIR}/scene_fixture.ts)
  set_tests_properties(scene_decoder PROPERTIES FIXTURES_REQUIRED scene_fixture)
endif()
ament_package()
